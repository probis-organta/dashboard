function parseCSV(text){
const rows=[]; let row=[], field='', i=0, inQ=false;
while(i<text.length){
const c=text[i];
if(inQ){ if(c==='"'){ if(text[i+1]==='"'){field+='"';i++;} else inQ=false; } else field+=c; }
else { if(c==='"') inQ=true; else if(c===','){ row.push(field); field=''; } else if(c==='\n'){ row.push(field); rows.push(row); row=[]; field=''; } else if(c==='\r'){} else field+=c; }
i++;
}
if(field.length>0||row.length>0){ row.push(field); rows.push(row); }
return rows;
}
function colOf(headersLower, aliases){
for(const a of aliases){ const i=headersLower.indexOf(a); if(i!==-1) return i; }
return -1;
}
function buildFromWide(rows){
const headers=rows[0];
const hl=headers.map(x=>String(x||'').trim().toLowerCase());
// resolve level columns
const levelCols=LEVELS.map(L=>({code:colOf(hl,L.code), name:colOf(hl,L.name)}));
if(levelCols[0].code===-1||levelCols[0].name===-1) throw {kind:'cols', headers};
// determine which level columns are present
const usedIdx=new Set();
levelCols.forEach(c=>{ if(c.code!==-1)usedIdx.add(c.code); if(c.name!==-1)usedIdx.add(c.name); });
// unit columns = any header not used as level col and not in NON_UNIT_HEADERS
const unitCols=[];
hl.forEach((h,idx)=>{ if(usedIdx.has(idx)) return; if(NON_UNIT_HEADERS.includes(h)) return; unitCols.push({idx, name:String(headers[idx]).trim()}); });
const map={}; const roots=[];
const data=rows.slice(1);
data.forEach(r=>{
let parentCode=null;
for(let lv=0; lv<levelCols.length; lv++){
const cc=levelCols[lv].code, nc=levelCols[lv].name;
if(cc===-1) break;
const code=String(r[cc]||'').trim();
const name=String(r[nc]||'').trim();
if(!code) break; // level belum ditentukan -> hentikan jalur baris ini
if(!map[code]){
const node={code,name,level:lv,children:[],units:[]};
map[code]=node;
if(parentCode && map[parentCode]) map[parentCode].children.push(node);
else if(lv===0) roots.push(node);
else if(parentCode==null) roots.push(node); // fallback
} else if(name && !map[code].name){ map[code].name=name; }
// attach units only at deepest filled level for this row (L3 typically)
parentCode=code;
// if this is the last level with data in the row, mark units
const nextHasData = (lv+1<levelCols.length && levelCols[lv+1].code!==-1 && String(r[levelCols[lv+1].code]||'').trim()!=='');
if(!nextHasData){
unitCols.forEach(u=>{ if(String(r[u.idx]||'').trim()==='1'){ if(!map[code].units.includes(u.name)) map[code].units.push(u.name); } });
break;
}
}
});
return roots;
}
function countLeaves(n){return n.children.length===0?1:n.children.reduce((s,c)=>s+countLeaves(c),0);}
function renderNode(node){
const hasKids=node.children.length>0;
const lv='lv'+Math.min(node.level,3);
let h='<li class="node '+lv+(hasKids?' collapsed':'')+'">';
h+='<div class="node-head"'+(hasKids?' onclick="tog(this)"':'')+'>';
h+='<span class="tw">'+(hasKids?'▶':'·')+'</span><span class="dot"></span>';
h+='<span class="code">'+esc(node.code)+'</span><span class="nm">'+esc(node.name)+'</span>';
if(hasKids){ h+='<span class="cnt">'+countLeaves(node)+(node.level===0?' proses':'')+'</span>'; }
else if(node.units && node.units.length){ node.units.forEach(u=>{const t=UNIT_TITLES[u.toUpperCase()]||u; h+='<span class="uchip" title="'+esc(t)+'">'+esc(u)+'</span>';}); }
h+='</div>';
if(hasKids) h+='<ul>'+node.children.map(renderNode).join('')+'</ul>';
h+='</li>';
return h;
}

async function loadData(){
setLive('stale','Memuat…');
showState('<div class="spin"></div><div class="big">Memuat data dari Google Spreadsheet…</div><div>Pastikan ada koneksi internet.</div>', false);

const roots=buildFromWide(rows);

}catch(err){

return;
}
const offline=(location.protocol==='file:');

else { msg+='<div style="margin-top:12px">Periksa: URL CSV benar &amp; sudah di-Publish to web, ada internet, Sheet masih dipublish.</div>'; }
showState(msg, true);
}
}

hits++; n.classList.add('hit'); n.style.display='';
